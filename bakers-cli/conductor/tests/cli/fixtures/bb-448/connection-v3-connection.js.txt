// @bakers-schema-version 3
// Canonical bakers.build CONNECTION config — committed (BB-201). A dropsh-shaped
// { site, plugins } read by 'bakers ui', 'bakers dropsh', and this conductor's auth.
// Connection + credentials come from your user-global machine context
// (~/.bakers/machine.config.js). The engine wiring lives in the sibling
// conductor.config.js (engine-only).
//
// plugins[] is consumed by DROPSH, whose resolver has no sole-function
// auto-pick, so @dropsh/plugin-oauth2 entries MUST name export: 'oauth2Plugin'.
// The '@bakers-build/addon-essentials' entry (BB-226) needs NO export: — that
// package default-exports its factory. It contributes the markdown renderer
// ('bakers dropsh --format md') plus the authoritative JSON:API schema.
async function loadMachine() {
  try {
    return (await import(`${process.env.HOME}/.bakers/machine.config.js`)).default ?? {};
  } catch {}
  return {};
}

async function loadLocal() {
  try { return (await import('./conductor.config.local.js')).default ?? {}; } catch {}
  return {};
}

const machine = await loadMachine();
const local = await loadLocal();
const baseUrl = local.base_url ?? machine.base_url;
const clientId = local.oauth?.client_id ?? machine.client_id ?? 'bakers-agent';
const clientSecret = local.oauth?.client_secret ?? machine.client_secret;
// BB-351: the ambient workspace every create is bound to. REQUIRED — see the
// top-level `workspace_id` below.
const workspaceId = local.workspace_id ?? machine.workspace_id;

export default {
  // BB-216: the connection-config schema version — kept in sync with the
  // header marker above so `bakers upgrade` can migrate a stale-shape config.
  schema_version: 3,
  site: { base_url: baseUrl, jsonapi_prefix: local.jsonapi_prefix ?? '/jsonapi' },
  // BB-351: REQUIRED, and top-level — a sibling of `site`, never a member of
  // it and never an addon's `with`. The filler that consumes it ships inside
  // `@bakers-build/addon-essentials`, which takes no options, so this is the only
  // place a connection ever names its workspace. `loadBbConfig` refuses a
  // config without it, so set `workspace_id` in ~/.bakers/machine.config.js (or
  // this repo's conductor.config.local.js) to the uuid of the bb_workspace
  // this checkout creates into.
  workspace_id: workspaceId,
  plugins: [
    { plugin: '@bakers-build/addon-essentials' },
    // BB-391: ONE identity, on ONE scope. There used to be two entries here
    // that differed only by the scope they requested, and every write had to
    // pick one — that choice is what went away. The id names the CONNECTION,
    // not a capability, which is why it is not named after a job.
    //
    // The scope is not ceremony: for a client_credentials token simple_oauth
    // overwrites the account's permissions with EXACTLY the scope's, so this is
    // the ceiling of what this identity may do. The rank it holds in a workspace
    // still decides WHICH rows it reaches.
    {
      plugin: '@dropsh/plugin-oauth2',
      export: 'oauth2Plugin',
      with: {
        id: 'bakers',
        default: true,
        type: 'oauth2_client_credentials',
        client_id: clientId,
        client_secret: clientSecret,
        token_url: `${baseUrl}/oauth/token`,
        scope: 'bb:session',
      },
    },
  ],
};
